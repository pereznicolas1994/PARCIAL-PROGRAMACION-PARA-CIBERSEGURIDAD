$lista = Get-Process
$cantidad = $lista.Length

[int]$limite = Read-Host "Ingresa el limite de entradas/procesos para el partido"

if ($cantidad -gt $limite -And $cantidad -lt 500) {
    Write-Host "Alerta: El servidor del partido supera el limite pero esta bajo los 500 procesos."
} else {
    Write-Host "El servidor funciona fuera del rango de alerta."
}
